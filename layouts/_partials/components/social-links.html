{{- $social := .Site.Params.social -}} {{- with $social -}} {{- $mastodon := strings.TrimSpace (.mastodon | default "") -}} {{- $mastodonParts := split (strings.TrimPrefix "@" $mastodon) "@" -}} {{- $mastodonIsValid := and (eq (len $mastodonParts) 2) (index $mastodonParts 0) (index $mastodonParts 1) -}} {{- $bluesky := strings.TrimSpace (.bluesky | default "") -}} {{- $x := strings.TrimPrefix "@" (strings.TrimSpace (.x | default "")) -}} {{- $instagram := strings.TrimPrefix "@" (strings.TrimSpace (.instagram | default "")) -}} {{- $rss := strings.TrimSpace (.rss | default "") -}} {{- if or $mastodonIsValid $bluesky $x $instagram $rss }} {{- if $mastodonIsValid }} Mastodon {{- end }} {{- with $bluesky }} Bluesky {{- end }} {{- with $x }} X {{- end }} {{- with $instagram }} Instagram {{- end }} {{- with $rss }} RSS {{- end }} {{- end }} {{- end }}
